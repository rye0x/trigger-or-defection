% --- Page geometry (single column) ---
\geometry{letterpaper, textwidth=6.5in, textheight=9in, top=1in, headsep=0.25in}

% --- Paragraphs & spacing ---
\setlength{\parindent}{0pt}
\setlength{\parskip}{5.5pt}
\linespread{1.05}                     % use \onehalfspacing for drafts
\setlist{itemsep=2pt, topsep=4pt}

% --- Section headings ---
\titleformat{\section}{\large\bfseries}{\thesection}{0.6em}{}
\titleformat{\subsection}{\normalsize\bfseries}{\thesubsection}{0.6em}{}
\titleformat{\subsubsection}{\normalsize\bfseries\itshape}{\thesubsubsection}{0.6em}{}
\titlespacing*{\section}{0pt}{1.4ex plus .4ex}{0.8ex}
\titlespacing*{\subsection}{0pt}{1.2ex plus .3ex}{0.6ex}

% --- Header / footer ---

\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\thepage}\renewcommand{\headrulewidth}{0pt}}

% --- Title block ---
\makeatletter
\renewcommand{\@maketitle}{%
  \newpage\null\vskip 0.5em
  \hrule height 2pt \vskip 0.25in
  \begin{center}{\LARGE\bfseries \@title\par}\end{center}
  \vskip 0.25in \hrule height 0.5pt \vskip 0.3in
  \begin{center}\lineskip .5em\begin{tabular}[t]{c}\@author\end{tabular}\end{center}
  \vskip 0.3in
}
\makeatother
